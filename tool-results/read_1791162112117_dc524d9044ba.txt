     1→"""API-CHAT-01 `POST /chat/completions` (§13.2, §13.7) and API-REQ-01 support.
     2→
     3→SSE wire format (§13.7, normative):
     4→- named event `mesh.meta` first (request_id, model, backend, queued_ms,
     5→  model_state);
     6→- unnamed events with OpenAI `chat.completion.chunk` JSON, terminated by the
     7→  literal `data: [DONE]`;
     8→- keepalive comment `: ping` every 15 s while queued/loading — including the
     9→  cold-load first-token wait (CI-21 mitigation "Pings + UI 'Loading model…'",
    10→  CI-18: the phone treats ≥ 45 s of silence as a dead stream);
    11→- the first-token TIME budget itself is the Backend adapter's §10.3 rule 4
    12→  (`first_token_timeout_seconds`, default 120 s) — it surfaces here as a
    13→  terminal `mesh.error` `BACKEND_TIMEOUT` (Appendix D 504), never as a raw
    14→  generator abort (NFR-REL-02);
    15→- named event `mesh.stats` last (before [DONE]);
    16→- `mesh.error` is terminal (no [DONE] follows);
    17→- Cancel (client close or API-REQ-01): abort upstream ≤ 1 s, free the slot,
    18→  no further events.
    19→
    20→The Agent is stateless for chat (ADR-011, FR-CHAT-04): nothing about Content
    21→is persisted or logged (canary test TC-SEC-01).
    22→
    23→M7 (FR-MM-01/02, §13.6): message content may be an OpenAI-style part list.
    24→- image parts pass through to the Backend and require the resolved model to
    25→  report the `vision` capability (§13.5) — else 501 UNSUPPORTED_CAPABILITY.
    26→- audio parts (`input_audio`) are ACCEPTED BY SHAPE but there is no v1
    27→  Backend contract for audio-in chat (§6: neither Backend documents it; S-13
    28→  routes STT through the separate Whisper service) — chat with audio parts
    29→  returns 501 UNSUPPORTED_CAPABILITY with a hint to use `POST /tasks`
    30→  type=transcribe (FR-MM-02).
    31→
    32→M8 (FR-RTE-01..03, §16.6): `model: "auto"` resolves via the §16.6 rule
    33→engine; the decision (winner + per-candidate scores) is explained in
    34→`mesh.meta.routing` (additive field, §13.10). FR-AGENT-RT (ADR-020): `tools`
    35→is 422 unless `[agent_runtime] enabled=true`; tools run through the bounded
    36→agent loop on the NON-STREAM path only in v1 (stream + tools → 422).

    74→def _capability_gate(messages: list[dict[str, Any]], entry: Any) -> None:
    75→    """M7 gate (§13.5 closed vocabulary; §10.4 reject-don't-guess):
    76→    image parts need `vision`; audio parts in chat are out of the v1 Backend
    77→    contract (S-13: STT lives in the separate Whisper adapter)."""
    78→    has_image = False
    79→    has_audio = False
    80→    for message in messages:
    81→        content = message.get("content")
    82→        if not isinstance(content, list):
    83→            continue
    84→        for part in content:
    85→            if not isinstance(part, dict):
    86→                continue
    87→            if part.get("type") == "image_url":
    88→                has_image = True
    89→            elif part.get("type") == "input_audio":
    90→                has_audio = True
    91→    if has_image and "vision" not in entry.capabilities:
    92→        raise MeshError(
    93→            "UNSUPPORTED_CAPABILITY",
    94→            "The selected model does not report vision (§13.5).",
    95→            details={"required_capability": "vision"},
    96→        )
    97→    if has_audio:
    98→        raise MeshError(
    99→            "UNSUPPORTED_CAPABILITY",
   100→            "Audio parts in chat are not supported in v1; use POST /tasks "
   101→            "type=transcribe (FR-MM-02, S-13).",
   102→            details={"required_capability": "speech_to_text", "hint": "POST /tasks"},
   103→        )

   106→def _required_from_messages(
   107→    messages: list[dict[str, Any]],
   108→) -> tuple[frozenset[str], frozenset[str]]:
   109→    """§16.6 required-set derivation [DESIGN mapping]: image parts → the
   110→    `image` modality + `vision` capability (§13.5 closed vocabulary)."""
   111→    modalities: set[str] = set()
   112→    capabilities: set[str] = set()
   113→    for message in messages:
   114→        content = message.get("content")
   115→        if not isinstance(content, list):
   116→            continue
   117→        for part in content:
   118→            if isinstance(part, dict) and part.get("type") == "image_url":
   119→                modalities.add("image")
   120→                capabilities.add("vision")
   121→    return frozenset(modalities), frozenset(capabilities)

   124→async def _resolve_model(
   125→    request: Request, payload: dict[str, Any]
   126→) -> tuple[Any, str, Any, RoutingDecision]:
   127→    """Resolve `model` — pinned id or §16.6 auto (M8). Returns the tuple the
   128→    handlers use plus the routing decision for mesh.meta."""
   129→    state = request.app.state
   130→    model_ref = str(payload["model"])
   131→    if model_ref != AUTO_MODEL:
   132→        backend, backend_model_id, entry = state.router.resolve(model_ref)
   133→        return (
   134→            backend,
   135→            backend_model_id,
   136→            entry,
   137→            RoutingDecision(
   138→                mode="pinned",
   139→                mesh_model_id=entry.mesh_model_id,
   140→                backend_id=backend.id,
   141→                reason="pinned by the request (§10.4)",
   142→            ),
   143→        )
   144→    modalities, capabilities = _required_from_messages(payload["messages"])
   145→    if payload.get("tools"):
   146→        capabilities = capabilities | {"tool_calling"}
   147→    # FR-RTE-03: the classifier (EXPERIMENTAL, off by default) may only
   148→    # REFINE the required set; on any failure the rule engine runs unchanged.
   149→    classifier = getattr(state, "classifier_fn", None)
   150→    if classifier is not None:
   151→        try:
   152→            future = classifier(payload["messages"])
   153→            if future is not None:
   154→                extra = await future  # FR-RTE-03 experimental path
   155→                if isinstance(extra, frozenset):
   156→                    capabilities = capabilities | extra
   157→        except Exception:  # noqa: BLE001 — experimental path degrades silently
   158→            pass
   159→    return state.router.resolve_auto(
   160→        messages=payload["messages"],
   161→        max_tokens=payload.get("max_tokens"),
   162→        required_modalities=modalities,
   163→        required_capabilities=capabilities,
   164→    )

   187→async def _run_with_tools(
   188→    request: Request,
   189→    backend: Any,
   190→    backend_model_id: str,
   191→    entry: Any,
   192→    payload: dict[str, Any],
   193→    job: Any,
   194→) -> Any:
   195→    """Bounded agent loop over ONE scheduler job (ADR-020)."""
   196→    from localmesh_agent.core.agent_loop import assistant_turn_from_chunk, run_agent_loop
   197→
   198→    state = request.app.state
   199→    tools = state.tool_registry
   200→    max_iterations = state.settings.agent_runtime.max_iterations
   201→
   202→    async def collect(back: Any, req: Any) -> Any:
   203→        text: str | None = None
   204→        finish: str | None = None
   205→        tool_calls: tuple[Any, ...] = ()
   206→        async for chunk in back.stream_chat(req, job.token):
   207→            if chunk.delta_content is not None:
   208→                text = (text or "") + chunk.delta_content
   209→            if chunk.finish_reason is not None:
   210→                finish = chunk.finish_reason

   212→                from localmesh_agent.core.agent_loop import parse_tool_calls
   213→
   214→                tool_calls = parse_tool_calls(list(chunk.tool_calls))

   218→        backend,
   219→        backend_model_id=backend_model_id,
   220→        mesh_model_id=entry.mesh_model_id,
   221→        messages=payload["messages"],
   222→        tools=tools,
   223→        max_iterations=max_iterations,
   224→        collect_fn=collect,
   225→        tools_wire=tuple(dict(tool) for tool in payload["tools"]),
   226→    )

   230→    """One §13.7 default event: OpenAI chat.completion.chunk JSON."""
   231→    delta: dict[str, str] = {}

   233→        delta["role"] = chunk.role
   234→    if chunk.delta_content is not None:
   235→        delta["content"] = chunk.delta_content

   237→        "id": f"chatcmpl-{request_id}",
   238→        "object": "chat.completion.chunk",

   246→async def _stream(
   247→    request: Request,
   248→    payload: dict[str, Any],
   249→    request_id: str,
   250→    job: Any,
   251→    backend: Any,
   252→    backend_model_id: str,
   253→    entry: Any,
   254→    routing: RoutingDecision | None = None,
   255→) -> AsyncIterator[str]:
   256→    state = request.app.state
   257→    scheduler = state.scheduler
   258→    store = state.store
   259→    metrics = getattr(state, "metrics", None)  # §20.1 (optional in tests)
   260→    mesh_model_id = entry.mesh_model_id
   261→    started_mono = time.monotonic()
   262→    store.append_audit(
   263→        "chat_started",
   264→        meta={"backend_id": backend.id, "mesh_model_id": mesh_model_id},
   265→    )
   266→
   267→    created = int(time.time())
   268→    ttft_ms: int | None = None
   269→    tokens_out = 0
   270→    backend_completion_tokens: int | None = None
   271→    finish_reason: str | None = None
   272→    job_released = False
   273→    cancelled = False
   274→    # Pending first-chunk fetch (ping race below); cancelled on any exit so a
   275→    # closed stream never leaks a task still pulling from the Backend.
   276→    pending_next: asyncio.Task[ChatChunk] | None = None
   277→
   278→    def release_job() -> None:
   279→        nonlocal job_released
   280→        if not job_released:
   281→            job_released = True
   282→            scheduler.release(job, duration_s=time.monotonic() - started_mono)
   283→
   284→    try:
   285→        # -- queued phase: pings every 15 s while waiting (§13.7) -------------
   286→        acquire_task = asyncio.create_task(scheduler.await_running(job))
   287→        try:
   288→            while not acquire_task.done():
   289→                done, _ = await asyncio.wait({acquire_task}, timeout=PING_INTERVAL_S)
   290→                if acquire_task not in done:
   291→                    yield ": ping\n\n"
   292→            await acquire_task  # raises CANCELLED if cancelled while queued
   293→        except BaseException:
   294→            release_job()
   295→            raise
   296→
   297→        # -- mesh.meta (first named event, §13.7) ------------------------------
   298→        meta = {
   299→            "request_id": request_id,
   300→            "model": mesh_model_id,
   301→            "backend": backend.id,
   302→            "queued_ms": job.queued_ms,
   303→            "model_state": entry.state,
   304→        }
   305→        if routing is not None:  # M8 (§16.6): additive field, §13.10
   306→            meta["routing"] = routing.to_meta()
   307→        yield _sse_event("mesh.meta", json.dumps(meta, separators=(",", ":"), ensure_ascii=False))
   308→
   309→        # -- generation: forward chunks, poll for client disconnect ------------
   310→        upstream = backend.stream_chat(
   311→            ChatRequest(
   312→                model=mesh_model_id,
   313→                backend_model_id=backend_model_id,
   314→                messages=tuple(_mesh_message(m) for m in payload["messages"]),
   315→                stream=True,
   316→                temperature=payload.get("temperature"),
   317→                top_p=payload.get("top_p"),
   318→                max_tokens=payload.get("max_tokens"),
   319→                stop=tuple(payload["stop"])
   320→                if isinstance(payload.get("stop"), list)
   321→                else (payload["stop"],)
   322→                if isinstance(payload.get("stop"), str)
   323→                else None,
   324→                presence_penalty=payload.get("presence_penalty"),
   325→                frequency_penalty=payload.get("frequency_penalty"),
   326→                seed=payload.get("seed"),
   327→            ),
   328→            job.token,
   329→        )
   330→        first_chunk = True
   331→        while True:
   332→            if await request.is_disconnected():
   333→                # Client closed: cancel upstream ≤ 1 s, no further events.
   334→                job.token.cancel()
   335→                cancelled = True
   336→                break
   337→            if job.token.cancelled:
   338→                # Revoke-during-stream (§15.6 / FR-PAIR-06): the scheduler set
   339→                # cancel_reason="device_revoked" — tell the phone best-effort,
   340→                # then close. Uniform non-revoke cancels stay silent.
   341→                if job.cancel_reason == "device_revoked":
   342→                    release_job()
   343→                    store.append_audit(
   344→                        "chat_finished",
   345→                        meta={
   346→                            "backend_id": backend.id,
   347→                            "mesh_model_id": mesh_model_id,
   348→                            "status": "failed",
   349→                        },
   350→                    )
   351→                    yield _sse_event(
   352→                        "mesh.error",
   353→                        json.dumps(
   354→                            {
   355→                                "error": {
   356→                                    "code": "DEVICE_REVOKED",
   357→                                    "message": "This device has been revoked.",
   358→                                    "retryable": False,
   359→                                    "request_id": request_id,
   360→                                    "details": {},
   361→                                }
   362→                            },
   363→                            separators=(",", ":"),
   364→                            ensure_ascii=False,
   365→                        ),
   366→                    )
   367→                    return
   368→                cancelled = True
   369→                break
   370→            try:
   371→                if first_chunk:
   372→                    # §13.7 "Model loading … then pings until first token" /
   373→                    # CI-21: race the first chunk against 15 s keepalives so a
   374→                    # cold model load (mesh.meta model_state="loading") never
   375→                    # looks dead to the phone's 45 s idle detector (CI-18).
   376→                    # The first-token TIME budget is the adapter's (§10.3 rule
   377→                    # 4, default 120 s): it raises MeshError("BACKEND_TIMEOUT")
   378→                    # through the task — caught below as a terminal mesh.error.
   379→                    # (A former hard `wait_for(…, 30)` here aborted the stream
   380→                    # with an uncaught TimeoutError — no mesh.error, no pings,
   381→                    # and 30 s ≠ the spec's 120 s cold-load budget.)
   382→                    pending_next = asyncio.ensure_future(upstream.__anext__())
   383→                    while True:
   384→                        done, _ = await asyncio.wait({pending_next}, timeout=PING_INTERVAL_S)
   385→                        if pending_next in done:
   386→                            break
   387→                        yield ": ping\n\n"
   388→                    chunk = pending_next.result()
   389→                    pending_next = None
   390→                    first_chunk = False
   391→                else:
   392→                    chunk = await upstream.__anext__()
   393→            except StopAsyncIteration:
   394→                break
   395→            except MeshError as error:
   396→                if error.code == "CANCELLED":
   397→                    # §15.6 / FR-PAIR-06: revoke-during-stream must surface
   398→                    # DEVICE_REVOKED (best effort) whether the loop noticed the
   399→                    # cancel between chunks (above) or the upstream abort
   400→                    # surfaced here.
   401→                    if job.cancel_reason == "device_revoked":
   402→                        release_job()
   403→                        store.append_audit(
   404→                            "chat_finished",
   405→                            meta={
   406→                                "backend_id": backend.id,
   407→                                "mesh_model_id": mesh_model_id,
   408→                                "status": "failed",
   409→                            },
   410→                        )
   411→                        yield _sse_event(
   412→                            "mesh.error",
   413→                            json.dumps(
   414→                                {
   415→                                    "error": {
   416→                                        "code": "DEVICE_REVOKED",
   417→                                        "message": "This device has been revoked.",
   418→                                        "retryable": False,
   419→                                        "request_id": request_id,
   420→                                        "details": {},
   421→                                    }
   422→                                },
   423→                                separators=(",", ":"),
   424→                                ensure_ascii=False,
   425→                            ),
   426→                        )
   427→                        return
   428→                    cancelled = True
   429→                    break
   430→                release_job()
   431→                yield _sse_event(
   432→                    "mesh.error",
   433→                    json.dumps(
   434→                        {
   435→                            "error": {
   436→                                "code": error.code,
   437→                                "message": error.message,
   438→                                "retryable": error.retryable,
   439→                                "request_id": request_id,
   440→                                "details": error.details or {},
   441→                            }
   442→                        },
   443→                        separators=(",", ":"),
   444→                        ensure_ascii=False,
   445→                    ),
   446→                )
   447→                store.append_audit(
   448→                    "chat_finished",
   449→                    meta={
   450→                        "backend_id": backend.id,
   451→                        "mesh_model_id": mesh_model_id,
   452→                        "status": "failed",
   453→                    },
   454→                )
   455→                if metrics is not None:
   456→                    metrics.inc("requests_total", labels={"status": "error", "backend": backend.id})
   457→                return
   458→            if ttft_ms is None and chunk.delta_content is not None:
   459→                ttft_ms = int((time.monotonic() - started_mono) * 1000)
   460→            if chunk.usage_completion_tokens is not None:
   461→                backend_completion_tokens = chunk.usage_completion_tokens
   462→            if chunk.finish_reason is not None:
   463→                finish_reason = chunk.finish_reason
   464→            if chunk.delta_content is not None:
   465→                tokens_out += 1
   466→            yield f"data: {_openai_chunk_json(request_id, mesh_model_id, chunk, created)}\n\n"
   467→
   468→        if cancelled:
   469→            release_job()
   470→            # §13.7 Cancel: abort upstream, free slot, NO further events.
   471→            if metrics is not None:
   472→                metrics.inc("requests_total", labels={"status": "cancelled", "backend": backend.id})
   473→            store.append_audit(
   474→                "chat_finished",
   475→                meta={
   476→                    "backend_id": backend.id,
   477→                    "mesh_model_id": mesh_model_id,
   478→                    "status": "cancelled",
   479→                },
   480→            )
   481→            return
   482→
   483→        # -- mesh.stats (last named event, before [DONE], §13.7) ---------------
   484→        duration_ms = int((time.monotonic() - started_mono) * 1000)
   485→        if backend_completion_tokens is not None:
   486→            tokens_out = backend_completion_tokens
   487→            token_source = "backend"
   488→        else:
   489→            token_source = "estimated"  # §13.7: estimate when usage absent
   490→        stats = MeshStats(
   491→            ttft_ms=ttft_ms if ttft_ms is not None else duration_ms,
   492→            tokens_out=tokens_out,
   493→            tokens_per_sec=round(tokens_out / (duration_ms / 1000), 2) if duration_ms > 0 else 0.0,
   494→            duration_ms=duration_ms,
   495→            finish_reason=finish_reason,
   496→            token_count_source=token_source,
   497→        )
   498→        yield _sse_event(
   499→            "mesh.stats",
   500→            json.dumps(stats.to_api_json(), separators=(",", ":"), ensure_ascii=False),
   501→        )
   502→        yield "data: [DONE]\n\n"
   503→        release_job()
   504→        if metrics is not None:
   505→            # §20.1 push families at the terminal event (Metadata only).
   506→            metrics.inc("requests_total", labels={"status": "ok", "backend": backend.id})
   507→            metrics.observe("ttft_ms", float(stats.ttft_ms))
   508→            if tokens_out > 0:
   509→                metrics.inc("tokens_out_total", labels={"backend": backend.id}, value=tokens_out)
   510→                metrics.set_gauge("tokens_per_sec", stats.tokens_per_sec)
   511→        store.append_audit(
   512→            "chat_finished",
   513→            meta={
   514→                "backend_id": backend.id,
   515→                "mesh_model_id": mesh_model_id,
   516→                "tokens_out": tokens_out,
   517→                "duration_ms": duration_ms,
   518→            },
   519→        )
   520→    finally:
   521→        if pending_next is not None and not pending_next.done():
   522→            pending_next.cancel()  # stop pulling from the Backend on exit
   523→        job.token.cancel()  # abort upstream on any generator exit (§13.7)
   524→        release_job()

   527→@router.post("/mesh/v1/chat/completions")
   528→async def chat_completions(request: Request) -> object:
   529→    principal = get_principal(request)
   530→    require_scope(principal, "chat")
   531→    try:
   532→        payload = await request.json()
   533→    except Exception as exc:  # malformed JSON body
   534→        raise MeshError("INVALID_REQUEST", "Request body must be valid JSON.") from exc
   535→    payload = validate_chat_payload(payload)
   536→    _tools_gate(request, payload)  # M8 (ADR-020): fail fast, before any work
   537→
   538→    request_id = request.state.mesh_request_id  # set by middleware (§13.3)
   539→
   540→    if payload.get("stream", True) is False:
