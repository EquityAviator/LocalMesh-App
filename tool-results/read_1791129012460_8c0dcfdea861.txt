   131→def _info_over_loopback(port: int) -> bool:
   132→    """True when something on 127.0.0.1:port answers `GET /mesh/v1/info` with
   133→    an agent-shaped payload (API-INFO-01). TLS first (§17.3 default), then
   134→    plain HTTP (§17.9 dev mode). Metadata only."""
   135→    context = ssl.create_default_context()
   136→    context.check_hostname = False
   137→    context.verify_mode = ssl.CERT_NONE  # self-signed identity is expected here
   138→    for scheme in ("https", "http"):
   139→        try:
   140→            with urllib.request.urlopen(  # noqa: S310 - loopback, scheme pinned above
   141→                f"{scheme}://127.0.0.1:{port}/mesh/v1/info", timeout=2, context=context
   142→            ) as response:
   143→                payload = json.loads(response.read().decode("utf-8") or "{}")
   144→            if isinstance(payload, dict) and "agent_id" in payload:
   145→                return True
   146→        except (urllib.error.URLError, OSError, ValueError):
   147→            continue
   148→    return False

   151→def _port_probe_default(host: str, port: int) -> PortState:
   152→    """Bind test → free; otherwise classify via the loopback /info probe."""
   153→    binder = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
   154→    try:
   155→        binder.bind((host, port))
   156→        return "free"
   157→    except OSError:
   158→        pass
   159→    finally:
   160→        binder.close()
   161→    return "listening" if _info_over_loopback(port) else "occupied"

   164→def _backend_probe_default(base_url: str, kind: str) -> tuple[str, str | None]:
   165→    """Loopback probe with the §6 native list endpoint (§10.3 probe)."""
   166→    import httpx
   167→
   168→    path = _BACKEND_PROBE_PATH.get(kind, "/v1/models")
   169→    try:
   170→        with httpx.Client(timeout=httpx.Timeout(3.0)) as client:
   171→            response = client.get(
   172→                base_url.rstrip("/") + path, headers={"Accept": "application/json"}
   173→            )
   174→    except httpx.HTTPError:
   175→        return "down", None
   176→    if response.status_code != 200:
   177→        return "down", None
   178→    return "up", None  # version stays None: no invented values (§13.5)

   203→def _mdns_browse_default() -> int | None:
   204→    """Best-effort zeroconf browse for `_localmesh._tcp` (2 s). Returns the
   205→    instance count, or None when multicast/zeroconf is unavailable."""
   206→    try:
   207→        from zeroconf.asyncio import AsyncServiceBrowser, AsyncZeroconf
   208→    except Exception:  # noqa: BLE001 - optional dependency path (T-21 hint)
   209→        return None
   210→
   211→    import asyncio
   212→
   213→    class _Counter:
   214→        def __init__(self) -> None:
   215→            self.count = 0
   216→
   217→        def add_service(self, _zc: Any, _type: str, _name: str) -> None:
   218→            self.count += 1
   219→
   220→        def update_service(self, _zc: Any, _type: str, _name: str) -> None:
   221→            pass
   222→
   223→        def remove_service(self, _zc: Any, _type: str, _name: str) -> None:
   224→            pass
   225→
   226→    async def _browse() -> int:
   227→        counter = _Counter()
   228→        aiozc = AsyncZeroconf()
   229→        try:
   230→            browser = AsyncServiceBrowser(
   231→                aiozc.zeroconf, "_localmesh._tcp.local.", listener=counter
   232→            )  # type: ignore[arg-type]
   233→            await asyncio.sleep(2.0)
   234→            await browser.async_cancel()
   235→        finally:
   236→            await aiozc.async_close()
   237→        return counter.count
   238→
   239→    try:
   240→        return asyncio.run(asyncio.wait_for(_browse(), timeout=4.0))
   241→    except Exception:  # noqa: BLE001 - multicast unavailable is a legal answer
   242→        return None

   245→def _firewall_probe_default(port: int) -> FirewallState:
   246→    """OS-specific best effort (§18.4): ufw on Linux; netsh on Windows;
   247→    socketfilterfw on macOS; anything else → unknown (manual step)."""
   248→    system = platform.system()
   249→    try:
   250→        if system == "Linux":
   251→            ufw = shutil.which("ufw")
   252→            if ufw is None:
   253→                return "unknown"
   254→            proc = subprocess.run(  # noqa: S603 - fixed argv, no shell
   255→                [ufw, "status"], capture_output=True, text=True, timeout=5, check=False
   256→            )
   257→            out = (proc.stdout + proc.stderr).lower()
   258→            if "inactive" in out:
   259→                return "unrestricted"  # no firewall: nothing blocks the port
   260→            if proc.returncode != 0:
   261→                return "unknown"  # typically needs root; manual step
   262→            return "rule-present" if f"{port}/tcp" in out and "allow" in out else "no-rule"
   263→        if system == "Windows":
   264→            netsh = shutil.which("netsh")
   265→            if netsh is None:
   266→                return "unknown"
   267→            proc = subprocess.run(  # noqa: S603 - fixed argv, no shell
   268→                [netsh, "advfirewall", "firewall", "show", "rule", "name=all"],
   269→                capture_output=True,
   270→                text=True,
   271→                timeout=10,
   272→                check=False,
   273→            )
   274→            return "rule-present" if str(port) in (proc.stdout or "") else "unknown"
   275→        if system == "Darwin":
   276→            fw = "/usr/libexec/ApplicationFirewall/socketfilterfw"
   277→            if not Path(fw).exists():
   278→                return "unknown"
   279→            proc = subprocess.run(  # noqa: S603 - fixed argv, no shell
   280→                [fw, "--getglobalstate"], capture_output=True, text=True, timeout=5, check=False
   281→            )
   282→            return "unknown" if proc.returncode != 0 else "unrestricted"
   283→    except (OSError, subprocess.TimeoutExpired):
   284→        return "unknown"
   285→    return "unknown"

   288→def _tailscale_probe_default() -> TailnetInfo | None:
   289→    """`tailscale status --json` [ASSUMPTION §6.3 fields]; None = absent."""
   290→    binary = shutil.which("tailscale")
   291→    if binary is None:
   292→        return None
   293→    try:
   294→        proc = subprocess.run(  # noqa: S603 - fixed argv, no shell
   295→            [binary, "status", "--json"], capture_output=True, text=True, timeout=5, check=False
   296→        )
   297→    except (OSError, subprocess.TimeoutExpired):
   298→        return TailnetInfo(state="unknown")
   299→    if proc.returncode != 0:
   300→        return TailnetInfo(state="unknown")
   301→    try:
   302→        payload = json.loads(proc.stdout)
   303→    except json.JSONDecodeError:
   304→        return TailnetInfo(state="unknown")
   305→    state = str(payload.get("BackendState") or "unknown")
   306→    self_obj = payload.get("Self") if isinstance(payload.get("Self"), dict) else {}
   307→    dns_name = self_obj.get("DNSName")
   308→    ips = tuple(str(ip) for ip in (self_obj.get("TailscaleIPs") or [])[:2])
   309→    return TailnetInfo(
   310→        state=state,
   311→        dns_name=str(dns_name).rstrip(".") if dns_name else None,
   312→        ips=ips,
   313→    )

   335→def _check_data_dir(settings: Settings) -> list[Finding]:
   336→    """§14.1: the data dir (and the TLS key inside it) is owner-only."""
   337→    path = settings.resolved_data_dir()
   338→    if not path.is_dir():
   339→        return [
   340→            Finding(
   341→                check="data_dir",
   342→                level="warn",
   343→                summary=f"Data dir {path} does not exist yet.",
   344→                detail="Created on first run with owner-only permissions (§14.1).",
   345→                fix="Run `localmesh-agent run` once, then re-run doctor.",
   346→            )
   347→        ]
   348→    findings: list[Finding] = []
   349→    if platform.system() != "Windows":
   350→        mode = stat.S_IMODE(path.stat().st_mode)
   351→        if mode & 0o077:
   352→            findings.append(
   353→                Finding(
   354→                    check="data_dir",
   355→                    level="warn",
   356→                    summary=f"Data dir {path} is readable by group/other (mode {mode:04o}).",
   357→                    detail="§14.1 requires owner-only permissions.",
   358→                    fix=f"`chmod 700 {path}`",
   359→                )
   360→            )
   361→            return findings
   362→    key_path = path / "tls" / KEY_FILENAME
   363→    if key_path.is_file() and platform.system() != "Windows":
   364→        key_mode = stat.S_IMODE(key_path.stat().st_mode)
   365→        if key_mode & 0o077:
   366→            findings.append(
   367→                Finding(
   368→                    check="data_dir",
   369→                    level="warn",
   370→                    summary=f"TLS private key {key_path} is group/other-readable "
   371→                    f"(mode {key_mode:04o}).",
   372→                    ci_ids=("CI-12",),  # exposed key can lead to identity theft → pin mismatch
   373→                    detail="§17.6: the private key file is owner-only (0600).",
   374→                    fix=f"`chmod 600 {key_path}`",
   375→                )
   376→            )
   377→    if not findings:
   378→        findings.append(
   379→            Finding(
   380→                check="data_dir",
   381→                level="ok",
   382→                summary=f"Data dir {path} perms owner-only (§14.1).",
   383→            )
   384→        )
   385→    return findings

   449→def _check_backends(settings: Settings, probes: DoctorProbes) -> list[Finding]:
   450→    """§18.4: 'Backends reachable on loopback'; CI-05 with per-kind fixes."""
   451→    findings: list[Finding] = []
   452→    for backend in settings.backends:
   453→        if not backend.enabled:
   454→            findings.append(
   455→                Finding(
   456→                    check="backends",
   457→                    level="info",
   458→                    summary=f"Backend '{backend.id}' is disabled in config (skipped).",
   459→                )
   460→            )
   461→            continue
   462→        status, _version = probes.backend_probe(backend.base_url, backend.kind)
   463→        if status == "up":
   464→            findings.append(
   465→                Finding(
   466→                    check="backends",
   467→                    level="ok",
   468→                    summary=f"Backend '{backend.id}' ({backend.kind}) up on loopback.",
   469→                )
   470→            )
   471→            continue
   472→        fix = (
   473→            "Start LM Studio and its local server (§18.2 CI-05)."
   474→            if backend.kind == "lmstudio"
   475→            else "Start Ollama (`ollama serve`) (§18.2 CI-05)."
   476→            if backend.kind == "ollama"
   477→            else "Start the OpenAI-compatible server (§18.2 CI-05)."
   478→        )
   479→        findings.append(
   480→            Finding(
   481→                check="backends",
   482→                level="warn",
   483→                summary=f"Backend '{backend.id}' ({backend.kind}) not reachable on "
   484→                f"{backend.base_url} (CI-05).",
   485→                ci_ids=("CI-05",),
   486→                fix=fix,
   487→            )
   488→        )
   489→    if not settings.backends:
   490→        findings.append(
   491→            Finding(
   492→                check="backends",
   493→                level="warn",
   494→                summary="No Backends configured — chat will return BACKEND_UNAVAILABLE.",
   495→                ci_ids=("CI-05",),
   496→                fix="Add an [[backends]] block to config.toml (Appendix E).",
   497→            )
   498→        )
   499→    return findings

   502→def _check_backend_bind(settings: Settings, probes: DoctorProbes) -> list[Finding]:
   503→    """§18.4: 'Backend bind-address warnings (CI-06/07)' — spec-named
   504→    detection: the Doctor probes the Backend on a LAN address (§18.2 Det)."""
   505→    lan_ip = probes.lan_ip()
   506→    if lan_ip is None:
   507→        return [
   508→            Finding(
   509→                check="backend_bind",
   510→                level="info",
   511→                summary="No LAN IPv4 (offline?) — CI-06/CI-07 bind checks skipped.",
   512→            )
   513→        ]
   514→    findings: list[Finding] = []
   515→    for backend in settings.backends:
   516→        if not backend.enabled or backend.kind not in ("lmstudio", "ollama"):
   517→            continue
   518→        port = backend.base_url.rsplit(":", 1)[-1]
   519→        if not port.isdigit():
   520→            continue
   521→        path = _BACKEND_PROBE_PATH[backend.kind]
   522→        if not probes.lan_backend_probe(lan_ip, int(port), path):
   523→            continue
   524→        if backend.kind == "lmstudio":
   525→            findings.append(
   526→                Finding(
   527→                    check="backend_bind",
   528→                    level="warn",
   529→                    summary=f"LM Studio answers on LAN address {lan_ip}:{port} (CI-06) — "
   530→                    "'serve on local network' is not needed (SEC-N2).",
   531→                    ci_ids=("CI-06",),
   532→                    detail="The Agent uses loopback only (§10.3).",
   533→                    fix="Turn LM Studio's local-network serving off.",
   534→                )
   535→            )
   536→        else:
   537→            findings.append(
   538→                Finding(
   539→                    check="backend_bind",
   540→                    level="warn",
   541→                    summary=f"Ollama answers on LAN address {lan_ip}:{port} without auth "
   542→                    f"(CI-07) — reachable beyond this machine.",
   543→                    ci_ids=("CI-07",),
   544→                    detail="OLLAMA_HOST is likely 0.0.0.0 (§18.2 CI-07).",
   545→                    fix="Rebind Ollama to 127.0.0.1 or firewall the port from the LAN.",
   546→                )
   547→            )
   548→    if not findings:
   549→        findings.append(
   550→            Finding(
   551→                check="backend_bind",
   552→                level="ok",
   553→                summary="No Backend exposure beyond loopback detected (SEC-N2).",
   554→            )
   555→        )
   556→    return findings

   559→def _check_mdns(settings: Settings, probes: DoctorProbes, agent_running: bool) -> list[Finding]:
   560→    """§18.4: 'mDNS advertisement visible on selected interfaces'. Hint-layer
   561→    only (T-21): unknown/absent multicast never blocks."""
   562→    if not settings.mdns.enabled:
   563→        return [
   564→            Finding(
   565→                check="mdns",
   566→                level="info",
   567→                summary="mDNS advertisement disabled in config ([mdns] enabled=false).",
   568→            )
   569→        ]
   570→    addresses = probes.mdns_addresses()
   571→    if not addresses:
   572→        return [
   573→            Finding(
   574→                check="mdns",
   575→                level="warn",
   576→                summary="No advertisable private IPv4 — LAN discovery will not find "
   577→                "this Agent (CI-24 pattern).",
   578→                ci_ids=("CI-24", "CI-01"),
   579→                detail="Virtual adapters are excluded by default (CI-24).",
   580→                fix="Connect a network, or set `[mdns] interfaces` in config.toml.",
   581→            )
   582→        ]
   583→    if not agent_running:
   584→        return [
   585→            Finding(
   586→                check="mdns",
   587→                level="info",
   588→                summary=f"Interface selection resolves {len(addresses)} address(es) "
   589→                f"({', '.join(addresses[:3])}); advertisement starts with the Agent.",
   590→            )
   591→        ]
   592→    seen = probes.mdns_browse()
   593→    if seen is None:
   594→        return [
   595→            Finding(
   596→                check="mdns",
   597→                level="info",
   598→                summary=f"Advertising on {len(addresses)} address(es); multicast browse "
   599→                "unavailable here (visibility unverified, T-21 hint layer).",
   600→            )
   601→        ]
   602→    if seen == 0:
   603→        return [
   604→            Finding(
   605→                check="mdns",
   606→                level="warn",
   607→                summary="Agent is running but no _localmesh._tcp instance is visible via "
   608→                "multicast (CI-01 pattern on this host).",
   609→                ci_ids=("CI-01",),
   610→                detail="Phones can still connect via stored endpoints/QR/Tailnet (§18.2 CI-01).",
   611→                fix="Check the network's multicast/AP-isolation settings (§18.2 CI-01).",
   612→            )
   613→        ]
   614→    return [
   615→        Finding(
   616→            check="mdns",
   617→            level="ok",
   618→            summary=f"mDNS advertisement visible ({seen} instance(s) browsed, "
   619→            f"{len(addresses)} advertised address(es)).",
   620→        )
   621→    ]

   624→def _check_firewall(settings: Settings, probes: DoctorProbes) -> list[Finding]:
   625→    """§18.4: 'firewall rule present for TCP 8443 (OS-specific, best effort)'."""
   626→    state = probes.firewall_probe(settings.listen.port)
   627→    if state == "rule-present":
   628→        return [
   629→            Finding(
   630→                check="firewall",
   631→                level="ok",
   632→                summary=f"Firewall allows inbound TCP {settings.listen.port} (§18.7).",
   633→            )
   634→        ]
   635→    if state == "unrestricted":
   636→        return [
   637→            Finding(
   638→                check="firewall",
   639→                level="ok",
   640→                summary="Firewall inactive/unrestricted — nothing blocks the port (§18.7).",
   641→            )
   642→        ]
   643→    if state == "no-rule":
   644→        return [
   645→            Finding(
   646→                check="firewall",
   647→                level="warn",
   648→                summary=f"No firewall rule allows inbound TCP {settings.listen.port} "
   649→                "(CI-03 pattern for LAN Phones).",
   650→                ci_ids=("CI-03",),
   651→                detail="Phones on this LAN would time out connecting (§18.2 CI-03).",
   652→                fix="Allow inbound TCP 8443 for the Private profile (§18.7), "
   653→                'e.g. `sudo ufw allow 8443/tcp`; set the network profile to "Private".',
   654→            )
   655→        ]
   656→    return [
   657→        Finding(
   658→            check="firewall",
   659→            level="info",
   660→            summary="Firewall state unknown here — verify a rule allows inbound TCP "
   661→            f"{settings.listen.port} (§18.7, manual step).",
   662→        )
   663→    ]

   666→def _check_tailscale(probes: DoctorProbes) -> list[Finding]:
   667→    """§18.4: 'Tailnet status (via tailscale CLI if present)' [ASSUMPTION
   668→    §6.3 fields]. Agent-side mapping: state ≠ Running → CI-15 pattern."""
   669→    info = probes.tailscale_probe()
   670→    if info is None:
   671→        return [
   672→            Finding(
   673→                check="tailscale",
   674→                level="info",

   676→                detail="Away-from-home use needs Tailscale on PC + Phone (§18.5 playbook 2).",
   677→            )
   678→        ]
   679→    if info.state == "Running":
   680→        where = info.dns_name or (info.ips[0] if info.ips else "tailnet")
   681→        return [

   683→                check="tailscale",
   684→                level="ok",
   685→                summary=f"Tailnet up — reachable as {where} (T2 endpoint, §6.3).",
   686→            )

   689→        return [
   690→            Finding(
   691→                check="tailscale",
   692→                level="info",
   693→                summary="Tailscale CLI present but its state is unreadable here "

   697→    return [
   698→        Finding(
   699→            check="tailscale",
   700→            level="warn",
   701→            summary=f"Tailscale present but not running (state: {info.state}) — "
   702→            "Phones cannot use T2 endpoints (CI-15 pattern).",

   709→def _check_clock(probes: DoctorProbes) -> list[Finding]:
   710→    """§18.4: 'clock sanity'. CI-23: token expiry is monotonic by design, so
   711→    wall-clock skew cannot break auth — but certs/audit use the wall clock."""
   712→    now = probes.wall_now()
   713→    if now.year < 2024:  # [DESIGN] plausibility floor for a 2026-dated spec
   714→        return [
   715→            Finding(
   716→                check="clock",
   717→                level="warn",
   718→                summary=f"System clock implausible ({now.isoformat()}) — certificate "
   719→                "validation and audit timestamps would be wrong.",
   720→                ci_ids=("CI-23",),
   721→                detail="Auth token expiry itself is monotonic-clock based (CI-23, by design).",
   722→                fix="Sync the system clock (NTP).",
   723→            )
   724→        ]
   725→    return [
   726→        Finding(
   727→            check="clock",
   728→            level="ok",
   729→            summary=f"Clock sanity ok (UTC {now.strftime('%Y-%m-%d')}); token expiry uses "

   735→def run_doctor(
   736→    settings: Settings | None,
   737→    probes: DoctorProbes | None = None,
   738→    *,
   739→    config_error: str | None = None,
   740→) -> list[Finding]:
   741→    """Run the §18.4 ladder in order and return findings in that order.
   742→
   743→    `config_error` short-circuits the ladder with the single `config`
   744→    finding when the settings could not even be loaded (the CLI passes the
   745→    exception text; §18.4 check 1) — `settings` may be None then.
   746→    """
   747→    if config_error is not None or settings is None:
   748→        return [
   749→            Finding(
   750→                check="config",
   751→                level="error",
   752→                summary=f"Configuration invalid: {config_error or 'no settings'}",
   753→                detail="The remaining checks need a valid config (§18.4 check 1).",
   754→                fix="Fix config.toml (Appendix E is the exact key contract).",
   755→            )
   756→        ]
   757→    probes = probes or default_probes(settings)
   758→
   759→    findings: list[Finding] = [
   760→        Finding(
   761→            check="config",
   762→            level="ok",
   763→            summary="Configuration valid (Appendix E key contract, strict).",
   764→        )
   765→    ]
   766→    findings += _check_data_dir(settings)
   767→    findings += _check_tls(settings)
   768→
   769→    port_findings = _check_port(settings, probes)
   770→    findings += port_findings
   771→    agent_running = any(
   772→        f.check == "port" and f.level == "ok" and "listening" in f.summary for f in port_findings
   773→    )
   774→
   775→    findings += _check_backends(settings, probes)
   776→    findings += _check_backend_bind(settings, probes)
   777→    findings += _check_mdns(settings, probes, agent_running)
   778→    findings += _check_firewall(settings, probes)
   779→    findings += _check_tailscale(probes)
   780→    findings += _check_clock(probes)
   781→
   782→    order = {name: i for i, name in enumerate(CHECK_ORDER)}
   783→    findings.sort(key=lambda f: order.get(f.check, len(order)))
   784→    return findings
   785→

   792→def worst_level(findings: list[Finding]) -> Level:
   793→    """Highest-severity level present (exit-code driver, [DESIGN])."""
   794→    worst: Level = "ok"
   795→    for finding in findings:
   796→        if _LEVEL_ORDER[finding.level] > _LEVEL_ORDER[worst]:
   797→            worst = finding.level
   798→    return worst

   801→def render_text(findings: list[Finding], *, header: str | None = None) -> str:
   802→    """Deterministic text output; findings keyed by CI-ID with fixes (§18.4).
   803→
   804→    `header` is printed verbatim when given (the CLI supplies the timestamped
   805→    line; golden tests pass a fixed one).
   806→    """
   807→    lines = [header] if header else []
   808→    if not header:
   809→        lines.append("LocalMesh Agent doctor — LM-ARCH-001 §18.4 connection ladder")
   810→    lines.append("")
   811→    for finding in findings:
   812→        marker = f"[{finding.level}]"
   813→        suffix = f" ({', '.join(finding.ci_ids)})" if finding.ci_ids else ""
   814→        lines.append(f"{marker:<8}{finding.check:<14}{finding.summary}{suffix}")
   815→        if finding.detail:
   816→            lines.append(f"{'':<8}{'':<14}{finding.detail}")
   817→        if finding.fix:
   818→            lines.append(f"{'':<8}{'':<14}fix: {finding.fix}")
   819→    worst = worst_level(findings)
   820→    lines.append("")
   821→    if worst == "ok":
   822→        lines.append("all checks ok")
   823→    else:
   824→        lines.append(f"worst finding level: {worst} ({len(findings)} finding(s))")
   825→    return "\n".join(lines)
